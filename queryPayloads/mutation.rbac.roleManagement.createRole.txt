mutation rbacRoleManagementCreateRole ( $accountId:ID! $roleManagementCreateRoleInput:RoleManagementCreateRoleInput!) {
	rbac ( accountId:$accountId ) {
		roleManagement {
			createRole ( input:$roleManagementCreateRoleInput   ) {
				role {
					id
					name
					description
					predefined
					applicableAccountType
					isUsedOnExternalAccess
					permission {
						resource
						action
					}
				}
			}
		}
	}	
}
