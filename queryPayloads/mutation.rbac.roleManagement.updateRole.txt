mutation rbacRoleManagementUpdateRole ( $accountId:ID! $roleManagementUpdateRoleInput:RoleManagementUpdateRoleInput!) {
	rbac ( accountId:$accountId ) {
		roleManagement {
			updateRole ( input:$roleManagementUpdateRoleInput   ) {
				role {
					id
					name
					description
					predefined
					applicableAccountType
					isUsedOnExternalAccess
					permission {
						resource
						action
					}
				}
			}
		}
	}	
}
