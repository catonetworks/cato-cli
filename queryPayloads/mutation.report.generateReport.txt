mutation reportGenerateReport ( $accountId:ID! $reportGenerateReportInput:ReportGenerateReportInput!) {
	report ( accountId:$accountId ) {
		generateReport ( input:$reportGenerateReportInput  )  {
			run {
				id
				reportConfigurationId
				reportName
				templateName
				runAt
				status
				trigger
				failureReason
				renderingFormat
				downloadLink {
					url
					expiresAt
				}
			}
		}
	}	
}
