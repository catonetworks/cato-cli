mutation reportingGenerateReport ( $accountId:ID! $reportingGenerateReportInput:ReportingGenerateReportInput!) {
	reporting ( accountId:$accountId ) {
		generateReport ( input:$reportingGenerateReportInput  )  {
			run {
				id
				reportConfigurationId
				reportName
				templateName
				runAt
				status
				trigger
				failureReason
				renderingFormat
				downloadLink {
					url
					expiresAt
				}
			}
		}
	}	
}
