mutation sitesAssignSocket ( $accountId:ID! $siteAssignSocketInput:SiteAssignSocketInput!) {
	sites ( accountId:$accountId ) {
		assignSocket ( input:$siteAssignSocketInput  )  {
			socket {
				id
				serial
				registrationStatus
				primary
				description
				site {
					id
					name
				}
			}
		}
	}	
}
