mutation userEnableUser ( $accountId:ID! $enableUserInput:EnableUserInput!) {
	user ( accountId:$accountId ) {
		enableUser ( input:$enableUserInput  )  {
			users {
				id
				userId
				objectId
				onPremSid
				firstName
				lastName
				userPrincipalName
				displayName
				phone
				email
				department
				jobTitle
				userStatus
				creationDate
				lastModified
				importType
				directoryId
				hasRemoteAccessEligibility
				authInfo {
					authMethod
					authStatus
				}
				riskScore
			}
		}
	}	
}
