query auditFeed ( $accountIDs:[ID!] $auditFieldFilterInput:[AuditFieldFilterInput!] $fieldNames:[AuditFieldName!] $marker:String $timeFrame:TimeFrame!) {
	auditFeed ( accountIDs:$accountIDs filters:$auditFieldFilterInput marker:$marker timeFrame:$timeFrame ) {
		accounts {
			id
			records ( fieldNames:$fieldNames   ) {
				account {
					description
					entity {
						id
						name
						type
					}
					helperFields
				}
				admin {
					id
					name
					type
				}
				apiKey {
					id
					name
					type
				}
				fields {
					name
					value {
						__typename
						... on DateValue {
							date
						}
						... on Entity {
							id
							name
							type
						}
						... on StringValue {
							string
						}
					}

				}
				fieldsMap
				flatFields
				object {
					id
					name
					type
				}
				time
			}
		}
		fetchedCount
		from
		hasMore
		marker
		to
	}	
}
