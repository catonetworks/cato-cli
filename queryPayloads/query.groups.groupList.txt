query groupsGroupList ( $accountId:ID! $groupListInput:GroupListInput) {
	groups ( accountId:$accountId ) {
		groupList ( input:$groupListInput  )  {
			items {
				id
				name
				description
				audit {
					updatedTime
					updatedBy {
						id
						name
					}
				}
				membersCount
				membersCountPerType {
					type
					membersCount
				}
			}
			paging {
				total
			}
		}
	}	
}
