query notificationIntegrationReferenceBodyTemplate ( $accountId:ID! $webhookTemplateRefInput:WebhookTemplateRefInput!) {
	notification ( accountId:$accountId ) {
		integration {
			referenceBodyTemplate ( input:$webhookTemplateRefInput   ) {
				id
				name
				bodyTemplate
				url
				requestMethod
				auth {
					authType
					... on BasicAuth {
						username
					}
					... on CustomHeaderAuth {
						name
					}
					... on CustomHeadersAuth {
						plainHeaders {
							name
							value
						}
						secretHeaders {
							name
						}
					}
				}
				header {
					name
					value
				}
				responseMapping {
					key
					value
				}
				webhookRelationType
			}
		}
	}	
}
