query notificationMailingListList ( $accountId:ID!) {
	notification ( accountId:$accountId ) {
		mailingListList {
			items {
				id
				name
				type
				address
				admin {
					id
					name
				}
				audit {
					updatedTime
					updatedBy {
						id
						name
					}
				}
			}
			paging {
				total
			}
		}
	}	
}
