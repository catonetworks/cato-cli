query rbacRoleManagementRoleList ( $accountId:ID! $roleManagementRoleListInput:RoleManagementRoleListInput!) {
	rbac ( accountId:$accountId ) {
		roleManagement {
			roleList ( input:$roleManagementRoleListInput   ) {
				items {
					id
					name
					description
					predefined
					applicableAccountType
					isUsedOnExternalAccess
					permission {
						resource
						action
					}
				}
				paging {
					total
				}
			}
		}
	}	
}
