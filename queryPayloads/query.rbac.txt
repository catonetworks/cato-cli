query rbac ( $accountId:ID! $id:ID! $roleManagementRoleListInput:RoleManagementRoleListInput!) {
	rbac ( accountId:$accountId ) {
		roleManagement {
			permissionCatalog {
				resource {
					resource
					supportedAction
				}
			}
			role ( id:$id   ) {
				id
				name
				description
				predefined
				isUsedOnExternalAccess
				permission {
					resource
					action
				}
			}
			roleList ( input:$roleManagementRoleListInput   ) {
				items {
					id
					name
					description
					predefined
					isUsedOnExternalAccess
					permission {
						resource
						action
					}
				}
				paging {
					total
				}
			}
		}
	}	
}
